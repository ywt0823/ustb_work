## 数据访问层

```java
@Repository
public interface SchedulePlanRepository extends JpaRepository<SchedulePlan, Long> {
    List<SchedulePlan> findByStatus(ScheduleStatus status);
    List<SchedulePlan> findByRoute(Route route);
    List<SchedulePlan> findByDriver(Driver driver);
}

@Repository
public interface RouteRepository extends JpaRepository<Route, Long> {
    List<Route> findByIsActiveTrue();
    Route findByRouteCode(String routeCode);
}

@Repository
public interface DriverRepository extends JpaRepository<Driver, Long> {
    List<Driver> findByIsActiveTrue();
    Driver findByDriverCode(String driverCode);
    List<Driver> findByWorkDaysLessThan(Integer maxWorkDays);
}

@Repository
public interface VehicleRepository extends JpaRepository<Vehicle, Long> {
    List<Vehicle> findByIsActiveTrue();
    Vehicle findByLicensePlate(String licensePlate);
    List<Vehicle> findByLastMaintenanceBefore(LocalDate date);
}

@Repository
public interface TicketRepository extends JpaRepository<Ticket, Long> {
    List<Ticket> findByRoute(Route route);
    List<Ticket> findByPurchaseTimeBetween(LocalDateTime start, LocalDateTime end);
    Ticket findByTicketCode(String ticketCode);
}
```

## 控制器层

```java
@RestController
@RequestMapping("/api/schedules")
public class ScheduleController {
    @Autowired
    private ScheduleService scheduleService;
    
    @PostMapping
    public ResponseEntity<SchedulePlan> createSchedulePlan(@RequestBody SchedulePlan plan);
    
    @PutMapping("/{id}")
    public ResponseEntity<SchedulePlan> updateSchedulePlan(@PathVariable Long id, @RequestBody SchedulePlan plan);
    
    @DeleteMapping("/{id}")
    public ResponseEntity<Void> deleteSchedulePlan(@PathVariable Long id);
    
    @GetMapping
    public ResponseEntity<List<SchedulePlan>> getSchedulePlans();
    
    @PostMapping("/{id}/copy")
    public ResponseEntity<SchedulePlan> copySchedulePlan(@PathVariable Long id, @RequestParam String newName);
}

@RestController
@RequestMapping("/api/routes")
public class RouteController {
    @Autowired
    private RouteService routeService;
    
    @PostMapping
    public ResponseEntity<Route> createRoute(@RequestBody Route route);
    
    @GetMapping
    public ResponseEntity<List<Route>> getRoutes();
    
    @GetMapping("/{id}/stations")
    public ResponseEntity<List<RouteStation>> getRouteStations(@PathVariable Long id);
}

@RestController
@RequestMapping("/api/drivers")
public class DriverController {
    @Autowired
    private DriverService driverService;
    
    @PostMapping
    public ResponseEntity<Driver> createDriver(@RequestBody Driver driver);
    
    @GetMapping
    public ResponseEntity<List<Driver>> getDrivers();
    
    @GetMapping("/available")
    public ResponseEntity<List<Driver>> getAvailableDrivers(@RequestParam LocalDateTime startTime, @RequestParam LocalDateTime endTime);
}
```

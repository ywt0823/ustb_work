# 机场巴士运营平台技术架构图

## 整体架构设计

### 1. 架构模式
- **前后端分离架构**

- **分层架构**：表现层、业务层、数据访问层、数据层

### 1. 调度服务 (Schedule Service)
```java
@SpringBootApplication
@EnableJpaRepositories
@EnableCaching
public class ScheduleServiceApplication {
    public static void main(String[] args) {
        SpringApplication.run(ScheduleServiceApplication.class, args);
    }
}

@RestController
@RequestMapping("/api/schedule")
@Validated
public class ScheduleController {
    
    @Autowired
    private ScheduleService scheduleService;
    
    @PostMapping("/plans")
    public ResponseEntity<SchedulePlan> createSchedulePlan(
            @Valid @RequestBody SchedulePlanDTO planDTO) {
        SchedulePlan plan = scheduleService.createSchedulePlan(planDTO);
        return ResponseEntity.ok(plan);
    }
    
    @GetMapping("/plans")
    public ResponseEntity<Page<SchedulePlan>> getSchedulePlans(
            @RequestParam(defaultValue = "0") int page,
            @RequestParam(defaultValue = "10") int size,
            @RequestParam(required = false) String status) {
        Page<SchedulePlan> plans = scheduleService.getSchedulePlans(page, size, status);
        return ResponseEntity.ok(plans);
    }
    
    @PostMapping("/plans/{id}/copy")
    public ResponseEntity<SchedulePlan> copySchedulePlan(
            @PathVariable Long id,
            @RequestParam String newName) {
        SchedulePlan copiedPlan = scheduleService.copySchedulePlan(id, newName);
        return ResponseEntity.ok(copiedPlan);
    }
}

@Service
@Transactional
public class ScheduleService {
    
    @Autowired
    private SchedulePlanRepository schedulePlanRepository;
    
    @Autowired
    private DriverRepository driverRepository;
    
    @Autowired
    private RouteRepository routeRepository;
    
    @Cacheable(value = "schedulePlans", key = "#id")
    public SchedulePlan getSchedulePlanById(Long id) {
        return schedulePlanRepository.findById(id)
                .orElseThrow(() -> new SchedulePlanNotFoundException("班次计划不存在"));
    }
    
    public SchedulePlan createSchedulePlan(SchedulePlanDTO planDTO) {
        SchedulePlan plan = new SchedulePlan();
        BeanUtils.copyProperties(planDTO, plan);
        
        // 验证司机和线路
        Driver driver = driverRepository.findById(planDTO.getDriverId())
                .orElseThrow(() -> new DriverNotFoundException("司机不存在"));
        Route route = routeRepository.findById(planDTO.getRouteId())
                .orElseThrow(() -> new RouteNotFoundException("线路不存在"));
        
        plan.setDriver(driver);
        plan.setRoute(route);
        plan.setStatus(ScheduleStatus.PENDING);
        
        return schedulePlanRepository.save(plan);
    }
    
    @CacheEvict(value = "schedulePlans", key = "#id")
    public SchedulePlan updateSchedulePlan(Long id, SchedulePlanDTO planDTO) {
        SchedulePlan existingPlan = getSchedulePlanById(id);
        BeanUtils.copyProperties(planDTO, existingPlan, "id");
        return schedulePlanRepository.save(existingPlan);
    }
}
```

### 2. 支付服务 (Payment Service)
```java
@Service
@Transactional
public class PaymentService {
    
    @Autowired
    private PaymentRepository paymentRepository;
    
    @Autowired
    private ThirdPartyPaymentClient thirdPartyPaymentClient;
    
    @Autowired
    private ICCardSystemClient icCardSystemClient;
    
    @Autowired
    private FinancialSystemClient financialSystemClient;
    
    public Payment processPayment(PaymentRequest request) {
        Payment payment = new Payment();
        payment.setAmount(request.getAmount());
        payment.setMethod(request.getMethod());
        payment.setStatus(PaymentStatus.PENDING);
        
        switch (request.getMethod()) {
            case ONLINE:
                return processOnlinePayment(payment, request);
            case IC_CARD:
                return processICCardPayment(payment, request);
            case TRANSFER:
                return processTransferPayment(payment, request);
            case CHECK:
                return processCheckPayment(payment, request);
            default:
                throw new UnsupportedPaymentMethodException("不支持的支付方式");
        }
    }
    
    @Async
    public Payment processOnlinePayment(Payment payment, PaymentRequest request) {
        try {
            ThirdPartyPaymentResponse response = thirdPartyPaymentClient
                    .processPayment(request.getThirdPartyRequest());
            
            if (response.isSuccess()) {
                payment.setStatus(PaymentStatus.COMPLETED);
                payment.setThirdPartyTransactionId(response.getTransactionId());
            } else {
                payment.setStatus(PaymentStatus.FAILED);
                payment.setRemark(response.getErrorMessage());
            }
            
            return paymentRepository.save(payment);
        } catch (Exception e) {
            payment.setStatus(PaymentStatus.FAILED);
            payment.setRemark(e.getMessage());
            return paymentRepository.save(payment);
        }
    }
    
    public Payment processICCardPayment(Payment payment, PaymentRequest request) {
        try {
            ICCardPaymentResponse response = icCardSystemClient
                    .processPayment(request.getIcCardRequest());
            
            if (response.isSuccess()) {
                payment.setStatus(PaymentStatus.COMPLETED);
                payment.setRemark("IC卡支付成功");
            } else {
                payment.setStatus(PaymentStatus.FAILED);
                payment.setRemark(response.getErrorMessage());
            }
            
            return paymentRepository.save(payment);
        } catch (Exception e) {
            payment.setStatus(PaymentStatus.FAILED);
            payment.setRemark(e.getMessage());
            return paymentRepository.save(payment);
        }
    }
    
    public Payment processTransferPayment(Payment payment, PaymentRequest request) {
        try {
            FinancialSystemResponse response = financialSystemClient
                    .processTransfer(request.getTransferRequest());
            
            if (response.isSuccess()) {
                payment.setStatus(PaymentStatus.COMPLETED);
                payment.setRemark("转账支付成功");
            } else {
                payment.setStatus(PaymentStatus.FAILED);
                payment.setRemark(response.getErrorMessage());
            }
            
            return paymentRepository.save(payment);
        } catch (Exception e) {
            payment.setStatus(PaymentStatus.FAILED);
            payment.setRemark(e.getMessage());
            return paymentRepository.save(payment);
        }
    }
}
```

### 3. 监控服务 (Monitor Service)
```java
@Service
public class MonitorService {
    
    @Autowired
    private GPSDataRepository gpsDataRepository;
    
    @Autowired
    private SafetyEventRepository safetyEventRepository;
    
    @Autowired
    private RabbitTemplate rabbitTemplate;
    
    @EventListener
    public void handleGPSDataReceived(GPSDataReceivedEvent event) {
        GPSData gpsData = event.getGpsData();
        
        // 存储GPS数据
        gpsDataRepository.save(gpsData);
        
        // 检查异常情况
        checkSafetyViolations(gpsData);
        
        // 发送到消息队列进行实时处理
        rabbitTemplate.convertAndSend("gps.data.queue", gpsData);
    }
    
    private void checkSafetyViolations(GPSData gpsData) {
        Vehicle vehicle = gpsData.getVehicle();
        
        // 检查超速
        if (gpsData.getSpeed() > vehicle.getMaxSpeed()) {
            createSafetyEvent(vehicle, SafetyEventType.SPEEDING, 
                    "车辆超速: " + gpsData.getSpeed() + " km/h");
        }
        
        // 检查偏离路线
        if (isOffRoute(gpsData)) {
            createSafetyEvent(vehicle, SafetyEventType.OFF_ROUTE, 
                    "车辆偏离预定路线");
        }
        
        // 检查长时间停车
        if (isLongTimeParking(gpsData)) {
            createSafetyEvent(vehicle, SafetyEventType.LONG_PARKING, 
                    "车辆长时间停车");
        }
    }
    
    private void createSafetyEvent(Vehicle vehicle, SafetyEventType type, String description) {
        SafetyEvent event = new SafetyEvent();
        event.setVehicle(vehicle);
        event.setType(type);
        event.setDescription(description);
        event.setEventTime(LocalDateTime.now());
        event.setLevel(SafetyLevel.WARNING);
        
        safetyEventRepository.save(event);
        
        // 发送通知
        rabbitTemplate.convertAndSend("safety.event.queue", event);
    }
}
```

## 性能优化

### 1. 缓存策略
```java
@Configuration
@EnableCaching
public class CacheConfig {
    
    @Bean
    public CacheManager cacheManager() {
        RedisCacheManager.Builder builder = RedisCacheManager
                .RedisCacheManagerBuilder
                .fromConnectionFactory(redisConnectionFactory())
                .cacheDefaults(cacheConfiguration());
        
        return builder.build();
    }
    
    private RedisCacheConfiguration cacheConfiguration() {
        return RedisCacheConfiguration.defaultCacheConfig()
                .entryTtl(Duration.ofMinutes(10))
                .serializeKeysWith(RedisSerializationContext.SerializationPair
                        .fromSerializer(new StringRedisSerializer()))
                .serializeValuesWith(RedisSerializationContext.SerializationPair
                        .fromSerializer(new GenericJackson2JsonRedisSerializer()));
    }
}
```
